function displayRoleOptions() {
    const roleList = document.getElementById('roleList');
    roleList.innerHTML = '';

    roleOptions.forEach(role => {
        const roleItem = document.createElement('div');
        roleItem.className = `role-item ${role.hasSelected ? 'disabled' : ''}`;

        const checkbox = document.createElement('input');
        checkbox.type = 'checkbox';
        checkbox.id = `role-${role.id}`;
        checkbox.className = 'role-checkbox';
        checkbox.value = role.id;
        checkbox.disabled = role.hasSelected;

        checkbox.addEventListener('change', updateSignButtonState);

        const label = document.createElement('label');
        label.htmlFor = `role-${role.id}`;
        label.textContent = role.name;

        if (role.hasSelected) {
            label.textContent += ' ' + t('alreadyInProcess');
        }

        roleItem.appendChild(checkbox);
        roleItem.appendChild(label);
        roleList.appendChild(roleItem);
    });
}

// 修改 addSign 函数中的错误提示
function addSign() {
    let requestData = {
        demandId: parseInt(demandId)
    };

    if(selectedTab === 'role'){
        const selectedRoles = [];
        const checkboxes = document.querySelectorAll('#roleList .role-checkbox:checked');
        checkboxes.forEach(checkbox => {
            selectedRoles.push(checkbox.value);
        })
        if(selectedRoles.length === 0){
            showError(t('selectRole'));
            return;
        }
        requestData.signType = 1;
        requestData.buRoleIds = selectedRoles.join(',');
    } else if (selectedTab === 'manual') {
        if(!window.selectedUser) {
            showError(t('selectUser'));
            return;
        }

        requestData.signType = 2;
        requestData.domainName = window.selectedUser.userName;
    }

    const signBtn = document.getElementById('signBtn');
    signBtn.disabled = true;
    signBtn.textContent = t('submitting');

    fetch('api/sign/submit',{
        method: 'POST',
        headers: {
            'Content-Type': 'application/json'
        },
        body: JSON.stringify(requestData)
    }).then(response => {
        if (response.status === 401) {
            redirectToAuth();
            return Promise.reject(new Error('UNAUTHORIZED'));
        }

        if (!response.ok) {
            throw new Error(`${t('submitFailed')} (${response.status})`);
        }
        return response.json();
    }).then(result => {
        if(result.msg && result.msg.trim() !== ''){
            showError(result.msg);
            loadRoleOptions();
            return;
        }

        if(result.success){
            showSuccess(t('submitSuccess'));
            clearSelection();
            loadRoleOptions();
        } else {
            showError(t('submitFailed') + ': ' + (result.msg || t('unknownError')));
            loadRoleOptions();
        }
        document.getElementById('searchInput').value = '';
        document.getElementById('clearBtn').style.display = 'none';
    }).catch(error => {
        if (!error.message.includes('UNAUTHORIZED')) {
            showError(t('requestFailed') + ': ' + error.message);
        }
    }).finally(()=>{
        signBtn.textContent = t('confirmBtn');
        updateSignButtonState()
    })
}

function updateLanguage() {
    const lang = i18n[currentLang];
    
    // 更新语言按钮文本
    document.getElementById('langBtn').textContent = currentLang === 'zh' ? 'EN' : '中文';
    
    // 更新页面标题
    document.querySelector('.page-header').textContent = lang.pageTitle;
    document.title = lang.pageTitle;
    
    // 更新Tab按钮
    const tabButtons = document.querySelectorAll('.tab-button');
    tabButtons[0].textContent = lang.roleTab;
    tabButtons[1].textContent = lang.manualTab;
    
    // 更新角色选择标题
    document.querySelector('.role-title').textContent = lang.roleSelectTitle;
    
    // 更新搜索输入框占位符
    document.getElementById('searchInput').placeholder = lang.searchPlaceholder;
    
    // 更新确认按钮
    const signBtn = document.getElementById('signBtn');
    if (signBtn.textContent !== lang.submitting && signBtn.textContent !== i18n.zh.submitting) {
        signBtn.textContent = lang.confirmBtn;

    // 更新模态框确认按钮

    // 重新显示角色选项(更新"已在审批流程"文本)
    if (roleOptions.length > 0) {
        displayRoleOptions();
    }
}
