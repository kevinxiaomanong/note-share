function addSign() {
    let requestData = {
        demandId: parseInt(demandId)
    };

    if (selectedTab === 'role') {
        // 角色选择模式
        const selectedRoles = [];
        const checkboxes = document.querySelectorAll('#roleList .role-checkbox:checked');
        checkboxes.forEach(checkbox => {
            selectedRoles.push(checkbox.value);
        });
    
        if (selectedRoles.length === 0) {
            showError('请选择至少一个审批角色');
            return;
        }
    
        requestData.signType = 1;
        requestData.buRoleIds = selectedRoles.join(',');
    
    } else if (selectedTab === 'manual') {
        // 手动输入模式
        if (!window.selectedUser) {
            showError('请先选择一个用户');
            return;
        }
    
        requestData.signType = 2;
        requestData.domainName = window.selectedUser.userName;
    }
    
    // 禁用按钮,防止重复提交
    const signBtn = document.getElementById('signBtn');
    signBtn.disabled = true;
    signBtn.textContent = '提交中...';
    
    // 调用后端接口
    fetch('/api/sign/submit', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json'
        },
        body: JSON.stringify(requestData)
    })
        .then(response => {
            if (response.status === 401) {
                redirectToAuth();
                return Promise.reject(new Error('UNAUTHORIZED'));
            }
    
            if (!response.ok) {
                throw new Error(`提交失败 (错误代码: ${response.status})`);
            }
            return response.json();
        })
        .then(result => {
            if (result.success) {
                showSuccess('加签申请提交成功!');
                // 清空选择状态
                clearSelection();
                // 重新加载角色选项(刷新"已在审批流程"状态)
                loadRoleOptions();
            } else {
                showError('加签申请提交失败: ' + (result.msg || '未知错误'));
            }
        })
        .catch(error => {
            if (!error.message.includes('UNAUTHORIZED')) {
                showError('加签申请提交失败: ' + error.message);
            }
        })
        .finally(() => {
            // 恢复按钮状态
            signBtn.textContent = '确认加签';
            updateSignButtonState();
        });
}

function getDemandDetail(id) {
    fetch(`/api/demand/detail?demandId=${id}`)
        .then(response => {
            if (response.status === 401) {
                redirectToAuth();
                return Promise.reject(new Error('UNAUTHORIZED'));
            }

            if (!response.ok) {
                throw new Error(`获取需求详情失败 (错误代码: ${response.status})`);
            }
            return response.json();
        })
        .then(result => {
            hideLoading();
            
            // 检查返回结果的成功状态
            if (!result.success) {
                showError(result.msg || '获取需求详情失败');
                return;
            }
            
            // 从 data 字段获取需求详情
            const data = result.data;
            if (data) {
                document.title = data.name || '需求详情';
                document.getElementById('pageHeader').textContent = data.name || '需求详情';

                const providerInfoDiv = document.getElementById('providerInfo');
                if (data.invalidDisplayName !== 1) {
                    const displayName = data.createByDisplayname || data.createBy || '-';
                    providerInfoDiv.textContent = `需求人:${displayName}`;
                } else {
                    providerInfoDiv.textContent = '';
                }
                providerInfoDiv.style.display = 'block';
                displayDemand(data.detail);
            } else {
                showError('未找到该需求信息');
            }
        })
        .catch(error => {
            hideLoading();

            if (error.message.includes('UNAUTHORIZED')) {
                return;
            }

            let errorMessage = '获取需求详情失败,请联系系统管理员S56025';
            if (error.message.includes('500')) {
                errorMessage = `查询需求${id}时出错,请联系系统管理员S56025`;
            }

            showError(errorMessage);
        });
}
